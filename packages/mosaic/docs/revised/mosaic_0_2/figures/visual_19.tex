\texttt{NodeContext} &

\texttt{NodeDecision} &
modelo: status, critérios com \texttt{criterionIndex}, satisfação, evidência em prosa e
\texttt{observationIds[]}, resultado, pedido de revisão e razão &
IDs opacos, não vazios, únicos por critério e causalmente disponíveis; não contém
observações nem \texttt{callId} \\
\hline
\texttt{Observation} &
runtime: \texttt{id}, \texttt{goalId}, \texttt{toolName}, \texttt{callId}, entrada, saída e ordem &
UUID opaco único no grafo ativo; chamada e retorno correlacionados exatamente uma vez;
\texttt{callId} não serve como referência;
a tool terminal de saída estruturada não é observação \\
\hline
\texttt{NodeOutcome} &
runtime: os campos validados de \texttt{NodeDecision}; observações pertencem ao \texttt{Node} &
referências são resolvidas contra o ledger local e a projeção causal do grafo ativo;
referências inválidas rejeitam o outcome antes de qualquer promoção \\
\hline
\texttt{RevisionRequest} &
modelo: \texttt{goalId}, \texttt{invalidatedAssumption}, \texttt{requestedEffect} &
\texttt{needs\_revision} exige pelo menos uma observação; o runtime associa
automaticamente o conjunto completo produzido pelo nó \\
\hline
\texttt{Artifact} &
runtime/modelo: \texttt{kind}, \texttt{mime} e \texttt{data} ou \texttt{reference} &
união estrita entre conteúdo inline e referência opaca; ordem preservada;
resolução e persistência fora do núcleo \\
\hline
\texttt{FinalDelivery} &
runtime: objetivos terminais, partes e artefatos discriminados &
construída sem nova chamada ao modelo; ordem estável e rastreável \\
